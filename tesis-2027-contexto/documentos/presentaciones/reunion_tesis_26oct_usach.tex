\begin{frame}{A1. Obs.\ 1--2: la ley anterior no cumple el objetivo}
  \footnotesize
  Ley evaluada: \;$\Pic{i}^{*}=\bar P_{\mathrm{net}}+\frac{E_i}{\tau}\big(\SOCw-\SOC{i}\big)$\quad$\Rightarrow$\quad
  $\sum_i E_i(\SOCw-\SOC{i})=0\;\Rightarrow\;\sum_i\PB{i}\equiv0$.

  \vspace{2pt}
  \begin{columns}[T,onlytextwidth]
    \column{.42\textwidth}
    \textbf{Obs.\ 1 (simulación):} $\max|\sum_i\PB{i}|=3\cdot10^{-15}$ kW y $\SOCw=0{,}5$ constante. Las BESS solo se intercambian energía entre sí, \textbf{a través de la red}.

    \vspace{4pt}
    \textbf{Obs.\ 2:} con netas sostenidas (3 / 1 / $-1$ kW), en régimen permanente $\Pic{i}\to\Pnet{i}$: igual que sin coordinación ($|I_n|=15{,}06$ A en ambos casos). Los SOC derivan a 0,1 / 0,5 / 0,9.
    \column{.56\textwidth}
    \centering
    \includegraphics[width=\linewidth]{obs2_step.pdf}
  \end{columns}
  \vspace{2pt}
  \begin{alertblock}{Conclusión}
    \footnotesize La ley es un \textbf{filtro pasa bajos} del desbalance, $1/(1+s\tau)$: atenúa transitorios, pero no reduce el desbalance sostenido. El $-42\,\%$ en el pico reportado antes provenía de filtrar la punta transitoria del EV.
  \end{alertblock}
\end{frame}

\begin{frame}{A2. Obs.\ 3: ¿cómo pasa energía entre fases sin desbalancear?}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
    \begin{block}{Resultado 1 (cabecera con tensión balanceada)}
      \footnotesize Con $\vec I_\phi=\overline{S_{h,\phi}}/\overline{V_\phi}$:
      \[ I_0=I_2=0\iff S_{h,a}=S_{h,b}=S_{h,c} \]
      $(I_0,I_1,I_2)$ es la DFT de $\overline{S_{h,\phi}}/V$: fuera de la componente continua solo se anula si las tres potencias complejas son iguales.
    \end{block}
    \footnotesize
    \textbf{Escenario original} (sin desbalance previo): transferir energía sostenida entre fases exige $P_{h,a}\ne P_{h,b}$ $\Rightarrow$ desbalance. \alert{Es incompatible}: la BESS de cada fase solo desplaza energía \emph{en el tiempo}.
    \column{.48\textwidth}
    \footnotesize
    \textbf{Escenario ampliado} (alimentador ya desbalanceado por clientes):
    \begin{itemize}\setlength\itemsep{2pt}
      \item La Micro-red de la fase cargada inyecta y la de la fase liviana absorbe: la transferencia ``a través de la red'' \textbf{iguala} $S_{h,\phi}$. Es compensación.
      \item Entre Micro-redes de la \textbf{misma fase} la transferencia es neutra para el desbalance.
      \item Límite energético: con $\Delta\mathrm{SOC}_\phi=0$ en el día,
      \[\textstyle\int\Pic{\phi}\,dt=\int(P_{PV}-P_{dl})_\phi\,dt\]
      la componente media diaria del desbalance \textbf{activo} queda fija por el diseño (piso de 5,1 A en el prototipo).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{B4. Modelo: balance DC y secuencias en cabecera}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
    \footnotesize
    \textbf{\color{usachT}Balance DC de cada Micro-red}
    \[\Pic{k}=P_{PV,k}-P_{cu,k}-P_{dl,k}+\PB{k}\]
    \[\SOC{k,t+1}=\SOC{k,t}-\PB{k,t}\,\Delta t/E_k\]
    \textbf{\color{usachT}Lo que ve la subestación}
    \[S_{h,\phi}=S_{L,\phi}-\textstyle\sum_{k\in\Nset{\phi}}S_{\mathrm{IC},k},\qquad \vec I_\phi=\overline{S_{h,\phi}}\,/\,\overline{V_\phi}\]
    \textbf{\color{usachT}Fortescue}
    \[\begin{bmatrix}I_0\\I_1\\I_2\end{bmatrix}=\frac13\begin{bmatrix}1&1&1\\1&a&a^2\\1&a^2&a\end{bmatrix}\begin{bmatrix}\vec I_a\\\vec I_b\\\vec I_c\end{bmatrix}\]
    \column{.5\textwidth}
    \footnotesize
    Con tensión balanceada en cabecera ($V_a=V$, $V_b=a^2V$, $V_c=aV$):
    \begin{block}{Definición de $B$ y $C$}
      \footnotesize
      \vspace{-4pt}
      \begin{gather*} B=\tfrac{1}{3V}\big(P_a+a^2P_b+aP_c\big)\\ C=\tfrac{1}{3V}\big(Q_a+a^2Q_b+aQ_c\big)\end{gather*}
      con $P_\phi=P_{h,\phi}$, $Q_\phi=Q_{h,\phi}$. Entonces
      \[ I_0=B-jC,\qquad I_2=\overline{B+jC}\;\Rightarrow\;|I_2|=|B+jC| \]
    \end{block}
    \begin{itemize}\setlength\itemsep{2pt}
      \item $B$ mide qué tan \textbf{distintas} son las $P$ de las tres fases; $C$, lo mismo con las $Q$.
      \item Si $P_a=P_b=P_c$, entonces $B=0$ (porque $1+a+a^2=0$).
      \item Todo es \textbf{lineal} en $P$ y $Q$: por eso el terciario es convexo.
    \end{itemize}
  \end{columns}
\end{frame}

    \end{tikzpicture}

    \vspace{2pt}
    {\scriptsize Las diagonales del paralelogramo son $I_0$ e $I_2$.}
    \column{.52\textwidth}
    \begin{block}{Ley del paralelogramo}
      \footnotesize
      \[ |I_0|^2+|I_2|^2=|B-jC|^2+|B+jC|^2=2|B|^2+2|C|^2 \]
    \end{block}
    \footnotesize
    \begin{itemize}\setlength\itemsep{1.5pt}
      \item $Q$ solo mueve $jC$: \textbf{no puede achicar $B$}; el mínimo de $|I_0|^2+|I_2|^2$ es $2|B|^2$.
      \item $Q$ sí \textbf{traslada} desbalance: si $jC=B$, $I_0=0$ pero $|I_2|=2|B|$ (ej.: $I_0$ 7,2$\to$0 A; $|I_2|$ 7,2$\to$14,4 A).
      \item Achicar $B$ exige que las $P$ por fase se parezcan, y eso cuesta \textbf{energía de las baterías} (SOC). $Q$ es ``gratis'' en energía, pero usa capacidad del IC: $Q^2\le S_{\max}^2-P^2$.
      \item \textcolor{usachR}{\textbf{Consecuencia de diseño:}} la parte media diaria de $B$ la fija \textbf{dónde y cuánta} FV/BESS hay en cada fase (piso energético).
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{B6. Nivel terciario: formulación (SOCP convexo)}
  \footnotesize
  \textbf{Variables:} $\Pic{\phi,t},\,\Qic{\phi,t},\,\PB{\phi,t},\,P_{cu,\phi,t},\,\SOC{\phi,t}$, \;$\phi\in\{a,b,c\}$, $t=1,\dots,H$
  \begin{align*}
    \min\;& \textstyle\sum_t\Big[c_t P_{h,t}+w_c\sum_\phi P_{cu,\phi,t}+w_0|I_{0,t}|^2+w_2|I_{2,t}|^2\Big]\Delta t\\
    \text{s.a.}\;& \Pic{\phi,t}=P_{PV,\phi,t}-P_{cu,\phi,t}-P_{dl,\phi,t}+\PB{\phi,t},\qquad 0\le P_{cu,\phi,t}\le P_{PV,\phi,t}\\
    & \SOC{\phi,t+1}=\SOC{\phi,t}-\PB{\phi,t}\Delta t/E_\phi,\quad \SOC{\phi,H}=\SOC{\phi,0},\quad \mathrm{SOC}^{\min}\le\SOC{\phi,t}\le\mathrm{SOC}^{\max}\\
    & |\PB{\phi,t}|\le P_{B,\phi}^{\max},\qquad \Pic{\phi,t}^2+\Qic{\phi,t}^2\le (S_\phi^{\max})^2\\
    & P_{h,t}\le P_{tr}^{\max},\qquad 3|I_{0,t}|\le I_n^{\max},\qquad |I_{2,t}|\le I_2^{\max}
  \end{align*}
  \vspace{-6pt}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
    \begin{itemize}\setlength\itemsep{2pt}
      \item Objetivo y restricciones convexos (normas de funciones afines de $P,Q$): \textbf{óptimo global}, resuelto con \texttt{CLARABEL}.
      \item Usa \textbf{agregados por fase}: el tamaño no crece con $N_\phi$.
    \end{itemize}
    \column{.48\textwidth}
    \begin{itemize}\setlength\itemsep{2pt}
      \item Horizonte deslizante (MPC económico): se reprograma cada 15 min.
      \item \textbf{Salida:} $P^*_{\phi,t},\,Q^*_{\phi,t}$ y la trayectoria $\SOC{\phi,t}$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{B8. Nivel secundario: reparto distribuido dentro de cada fase}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{.56\textwidth}
    \textbf{Problema por fase $\phi$} (cada periodo de control):
    \begin{align*}
      \min_{\PB{k}}\;&\textstyle\sum_{k\in\Nset{\phi}} C_k(\PB{k}),\quad C_k=\tfrac{a_k}{2}\PB{k}^2+b_k(\SOC{k})\,\PB{k}\\
      \text{s.a.}\;&\textstyle\sum_{k\in\Nset{\phi}}\Pic{k}=P^*_\phi+\Delta P_\phi^{\mathrm{corr}},\qquad |\PB{k}|\le P_{B,k}^{\max}
    \end{align*}
    {\scriptsize $a_k=1/E_k$; \; $b_k=\beta(\mathrm{SOC}^{\mathrm{ref}}-\SOC{k})$: descargar con SOC bajo es caro $\Rightarrow$ balance de SOC \textbf{dentro de la fase}.}

    \vspace{3pt}
    \textbf{KKT:} $a_k\PB{k}+b_k=\lambda_\phi$ para todo $k$ no saturado $\Rightarrow$ \alert{consenso en $\lambda_\phi$} (cap.\ 4 de [2], con un $\lambda$ por fase).

    \vspace{3pt}
    \textbf{Algoritmo} [15], grafo $\mathcal G_\phi$, pesos $w_{kj}$:
    \begin{align*}
      \lambda_k&\leftarrow\textstyle\sum_j w_{kj}\lambda_j+\varepsilon\,y_k\\
      \PB{k}&\leftarrow\arg\min\,\{C_k(\PB{k})-\lambda_k\PB{k}\}\\
      y_k&\leftarrow\textstyle\sum_j w_{kj}y_j-(\PB{k}^{\mathrm{new}}-\PB{k}^{\mathrm{old}})
    \end{align*}
    \column{.42\textwidth}
    \begin{itemize}\setlength\itemsep{3pt}
      \item $y_k$ estima de forma distribuida el desbalance de potencia de la fase; solo el \textbf{líder} conoce $P^*_\phi$ y la medición de cabecera ($\Delta P_\phi^{\mathrm{corr}}$ corrige el error de pronóstico).
      \item \textbf{Reactivos:} $\sum_k\Qic{k}=Q^*_\phi$ con consenso en la razón de uso $\Qic{k}/\sqrt{(S_k^{\max})^2-\Pic{k}^2}$.
      \item Diferencia con el trabajo previo: el consenso ya no decide el balance entre fases; lo \textbf{ejecuta} dentro de cada fase.
    \end{itemize}
  \end{columns}
\end{frame}
