\begin{frame}{El lazo rápido es una política afín de reserva}
  \vspace{-0.30cm}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
    \begin{tikzpicture}\node[kc=cS,text width=5.75cm]{%
      \kt{usachTd}{Política afín de reserva [Warrington et al., 2013]}\\[3pt]
      \hspace*{6pt}$\displaystyle u_i(t)=\underbrace{e_i(t)}_{\text{nominal}}+\sum_{s\le t}D_i(t,s)\,\delta(s)$\\[3pt]
      $\bullet$ $e_i$: programa que se fija antes (el plan).\\
      $\bullet$ $D_i$: respuesta planificada al error $\delta$ que se va revelando; triangular inferior (causal).\\
      $\bullet$ $D_i$ \textbf{solo diagonal} $=$ reserva secundaria clásica: el \textbf{factor de participación} del controlador.\\[2pt]
      };\end{tikzpicture}\\[4pt]
    \begin{tikzpicture}\node[kc=cL,text width=5.75cm]{%
      \kt{verdeL}{El lazo de cada fase es ese caso diagonal}\\[3pt]
      \hspace*{6pt}$P_{\mathrm{IC},k}(t)=P^*_k(t)+\alpha_k\,\Delta P_\phi(t),\qquad \Delta P_\phi\approx e_\phi$\\[3pt]
      $\bullet$ nominal $e_i$ \;$\leftrightarrow$\; $P^*_k$ (lo fija el terciario)\\
      $\bullet$ diagonal $D_i$ \;$\leftrightarrow$\; $\alpha_k\ge0$, $\sum_{k\in\mathcal N_\phi}\alpha_k=1$\\
      $\bullet$ error $\delta$ \;$\leftrightarrow$\; $e_\phi=P_{h,\phi}-\bar P_h$, medido en la cabecera\\[2pt]
      };\end{tikzpicture}

    \column{.51\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm,font=\fontsize{6.4}{7.6}\selectfont]
      \node[blq=gray!60!black,text width=3.6cm,font=\footnotesize] (E) at (3.1,4.15) {lo que verá el lazo en la fase $\phi$\\[1pt] $e_\phi(t)=\hat e_\phi(t)+\xi_\phi(t)$};
      \node[blq=cT,text width=2.75cm,align=left,font=\fontsize{6.4}{7.6}\selectfont] (A) at (1.45,2.05) {%
        \textbf{$\hat e_\phi$: previsible}\\ perfil conocido, dura horas\\[2pt]

      \node[blq=cP,text width=2.75cm,align=left,font=\fontsize{6.4}{7.6}\selectfont] (B) at (4.75,2.05) {%
        \textbf{$\xi_\phi$: error}\\ del pronóstico, media cero\\[2pt]

\begin{frame}{Modelo del error: de dónde salen $\sigma_\phi$, $z$ y $m$}
  \vspace{-0.32cm}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
    \begin{tikzpicture}\node[kc=cS,text width=5.75cm]{%
      \kt{usachTd}{1. Pronóstico simple y su error}\\[2pt]
      Demanda no gestionada de la fase, respecto del promedio:\\[1pt]
      \hspace*{6pt}$D_\phi=\big(P_{h,\phi}+\textstyle\sum_{k\in\mathcal N_\phi}P_{\mathrm{IC},k}\big)-\tfrac13\sum_\psi(\cdot)_\psi$\\[2pt]
      Pronóstico: mismo tramo de 15 min de los últimos $n$ días\\[1pt]
      \hspace*{6pt}$\hat D_\phi(t)=\tfrac1n\sum_{d=1}^{n}D_\phi(t-96d)$\\[2pt]
      Error: $\xi_\phi=D_\phi-\hat D_\phi$\ \ [kW], con $\sigma_\phi(h)$ \textbf{por hora del día} y autocorrelación $a_\phi$ entre tramos. Como $\sum_\phi\xi_\phi=0$, basta estimar dos fases.\\[2pt]
      };\end{tikzpicture}\\[4pt]
    \begin{tikzpicture}\node[kc=grisM,text width=5.75cm]{%
      \kt{grisM}{3. Qué cubre $m$}\\[2pt]
      Lo que el promedio de 15 min no ve: variación dentro del tramo, error de medida y de seguimiento del PI:\\[1pt]
      \hspace*{6pt}$m=q_{1-\varepsilon}\big(\,|P^{1\,\mathrm{min}}_{h,\phi}-\bar P^{15\,\mathrm{min}}_{h,\phi}|\,\big)$\\[2pt]
      };\end{tikzpicture}

    \column{.51\textwidth}
    \begin{tikzpicture}\node[kc=cT,text width=6.3cm]{%
      \kt{cT!85!black}{2. Margen de la fase: restricción probabilística}\\[2pt]
      \hspace*{6pt}$\Pr\big[\xi_\phi\le R^{\varepsilon\uparrow}_\phi\big]\ge1-\varepsilon,\qquad \Pr\big[\xi_\phi\ge -R^{\varepsilon\downarrow}_\phi\big]\ge1-\varepsilon$\\[2pt]
      \hspace*{6pt}$\Rightarrow\ R^{\varepsilon\uparrow}_\phi=q_{1-\varepsilon}(\xi_\phi)+m,\qquad R^{\varepsilon\downarrow}_\phi=-q_{\varepsilon}(\xi_\phi)+m$\\[1pt]
      };\end{tikzpicture}\\[2pt]
    \centering
    \begin{tikzpicture}[x=0.62cm,y=0.85cm,font=\fontsize{5.8}{6.4}\selectfont]

      \node at (0,-0.22) {0};
    \end{tikzpicture}\\[2pt]
    \raggedright
    \begin{tikzpicture}\node[kc=cT,text width=6.3cm]{%
      \kt{cT!85!black}{Cómo se calcula $q_{1-\varepsilon}$}\\[2pt]
      \begin{tabular}{@{}lcc@{}}
        & $\varepsilon=5\,\%$ & $\varepsilon=1\,\%$\\\hline

  \end{columns}
\end{frame}

      \hspace*{6pt}$\hat y_k=\alpha_k\,b_\phi$\quad (energía comprometida)\\[1pt]

      };\end{tikzpicture}

    \column{.54\textwidth}

      \hspace*{6pt}$\displaystyle \alpha_k=\frac{E_k}{\sum_{j\in\mathcal N_\phi}E_j}\ \Rightarrow\ \Delta\mathrm{SoC}_k=\frac{\alpha_k\,\Delta E_\phi}{E_k}=\frac{\Delta E_\phi}{\sum_j E_j}$\\[2pt]
      Todas las MR de la fase gastan \textbf{el mismo $\Delta$SoC}: nadie se agota antes que las demás. $E_k$: energía útil de BESS y EV (la informa el secundario).\\[1pt]
      };\end{tikzpicture}\\[4pt]
    \begin{tikzpicture}\node[kc=grisM,text width=6.55cm]{%

      \end{tabular}\\[2pt]
      \raggedright
      {\color{grisM}$R^{\varepsilon}_a=z\sigma_a+m$ con $\sigma_a=0{,}5$ kW, $z=2{,}33$, $m=0{,}2$ kW (supuestos).}\\[1pt]
      };\end{tikzpicture}\\[4pt]

    \column{.56\textwidth}
    \begin{tikzpicture}\node[kc=cT,text width=7.0cm]{%
      \kt{cT!85!black}{Objetivo: solo energía}\\[2pt]
      \hspace*{6pt}$\displaystyle\min\ \sum_{t}c(t)\,g(t)\,\Delta t$\\[3pt]
      \kt{cT!85!black}{Definiciones (lineales en $P^*$ y $\rho$)}\\[1pt]
      \hspace*{6pt}$S_\phi=\sum_{k\in\mathcal N_\phi}P^*_k,\quad \bar S=\tfrac13\sum_\psi S_\psi,\quad \hat e_\phi=\hat D_\phi-(S_\phi-\bar S)$\\[1pt]
      \hspace*{6pt}$b_\phi=\hat e_\phi-\rho_\phi,\quad \hat y_k=\alpha_k\,b_\phi,\quad \tilde P_k=P^*_k+\hat y_k$\\[3pt]
      \kt{cT!85!black}{Restricciones, para todo $t$, $\phi$ y $k\in\mathcal N_\phi$}\\[1pt]
      \textbf{(0)} compra:\ \ $g\ge\hat P_{L,\mathrm{tot}}-\sum_k\tilde P_k,\quad g\ge0$\\[2pt]
      \textbf{(a)} residual:\ \ $|\rho_\phi|\le\bar\rho,\quad \sum_\phi\rho_\phi=0$\\[2pt]
      \textbf{(b)} potencia:\ \ $P^{\min}_k+\alpha_kR^{\varepsilon\downarrow}_\phi\le\tilde P_k\le P^{\max}_k-\alpha_kR^{\varepsilon\uparrow}_\phi$\\[2pt]
      \textbf{(c)} energía:\ \ $W^{\min}_k(t)+\alpha_kZ^{\downarrow}_\phi(t)\le\sum_{s\le t}\tilde P_k(s)\Delta t\le W^{\max}_k(t)-\alpha_kZ^{\uparrow}_\phi(t)$\\[2pt]
      \textbf{(d)} margen de la fase:\ \ $\sum_k r_k=R^{\varepsilon}_\phi$, se cumple por construcción\\[2pt]
      \textbf{(e)} final del horizonte:\ \ $\sum_{s<T}\tilde P_k(s)\Delta t\le W^{\mathrm{fin}}_k$\\[2pt]
      };\end{tikzpicture}\\[4pt]
    \begin{tikzpicture}\node[kc=grisM,text width=7.0cm]{%
      \kt{grisM}{Qué agrega esta formulación}\\[1pt]
      $\bullet$ La compra usa $\tilde P_k$, no $P^*_k$; con $\sum_\phi\rho_\phi=0$ el balance previsto no mueve la compra total.\\
      $\bullet$ (c) usa $Z_\phi(t)$, que crece con el horizonte, en vez de $r\tau$ constante.\\
      $\bullet$ (e) evita que el LP vacíe las baterías al final de las 12 h.\\[1pt]
      };\end{tikzpicture}
