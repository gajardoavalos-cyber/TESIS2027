    \node[card=5.6cm] (d) at (6.2,-1.75) {\textbf{\color{usachT}FV y EV monofásicos}\\ Se conectan donde cae el cliente, no donde conviene: el desbalance crece y cambia en el día.};

  \footnotesize\raggedright
  \textbf{\color{usachO}Cómo se enfrenta hoy}
  \begin{itemize}\setlength\itemsep{1pt}
    \item Reconexión de clientes entre fases: manual, discreta, no sigue la variación diaria.
    \item Compensadores (STATCOM, filtros activos): inversión adicional; con solo $Q$ no se corrige el desbalance de potencia activa. \resp{r2}{R2}
  \end{itemize}
  \textbf{Oportunidad:} las Micro-redes DC con batería \textbf{ya están} en cada fase y pueden mover $P$ y $Q$.

\begin{frame}{3. Pregunta de investigación y objetivo general}
  \vspace{0.3cm}
  \begin{block}{Pregunta de investigación}
    \footnotesize ¿Cómo coordinar varias Micro-redes DC monofásicas por fase para gestionar el intercambio de energía y el desbalance en la cabecera de un alimentador BT 3$\phi$-4W, respetando la energía disponible en cada fase?
  \end{block}
  \vspace{0.4cm}
  \begin{block}{Objetivo general}
    \footnotesize Diseñar y validar, en simulación offline y en tiempo real, una estrategia de control jerárquica (terciario centralizado en cabecera + secundario distribuido por fase) que use $P$ y $Q$ de los IC de varias Micro-redes DC por fase para gestionar el intercambio de energía y el desbalance en la cabecera de un alimentador BT 3$\phi$-4W.
  \end{block}
\end{frame}

\begin{frame}{4. Formulación del problema del terciario}
  \footnotesize
  \textbf{Variables:} $\Pic{\phi,t},\,\Qic{\phi,t},\,\PB{\phi,t},\,P_{cu,\phi,t},\,\SOC{\phi,t}$, \;$\phi\in\{a,b,c\}$, $t=1,\dots,H$
  \begin{align*}
    \min\;& \textstyle\sum_t\Big[c_t P_{h,t}+w_c\sum_\phi P_{cu,\phi,t}+w_0|I_{0,t}|^2+w_2|I_{2,t}|^2\Big]\Delta t\\
    \text{s.a.}\;& \Pic{\phi,t}=P_{PV,\phi,t}-P_{cu,\phi,t}-P_{dl,\phi,t}+\PB{\phi,t},\qquad 0\le P_{cu,\phi,t}\le P_{PV,\phi,t}\\
    & \SOC{\phi,t+1}=\SOC{\phi,t}-\PB{\phi,t}\Delta t/E_\phi,\quad \SOC{\phi,H}=\SOC{\phi,0},\quad \mathrm{SOC}^{\min}\le\SOC{\phi,t}\le\mathrm{SOC}^{\max}\\
    & |\PB{\phi,t}|\le P_{B,\phi}^{\max},\qquad \Pic{\phi,t}^2+\Qic{\phi,t}^2\le (S_\phi^{\max})^2\\
    & P_{h,t}\le P_{tr}^{\max},\qquad 3|I_{0,t}|\le I_n^{\max},\qquad |I_{2,t}|\le I_2^{\max}
  \end{align*}
  \vspace{-4pt}
  \begin{itemize}\setlength\itemsep{2pt}
    \item $I_0$, $I_2$ son afines en $(P,Q)$ (R1) $\Rightarrow$ objetivo cuadrático y restricciones de cono: \textbf{SOCP}, óptimo global.
    \item Sin variables binarias (no hay diésel ni encendido/apagado), a diferencia de los EMS tipo MILP [8].
    \item En MPC deslizante, $\SOC{\phi,H}=\SOC{\phi,0}$ se reemplaza por costo terminal o referencia de SOC.
  \end{itemize}
  \end{frame}

\begin{frame}[label=r1]{R1. Derivación de $B$, $C$ y del paralelogramo}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
    Cabecera balanceada: $V_a=V$, $V_b=a^2V$, $V_c=aV$, \;$a=1\angle120^\circ$.
    \[ \vec I_a=\tfrac{P_a-jQ_a}{V},\;\; \vec I_b=\tfrac{(P_b-jQ_b)\,a^2}{V},\;\; \vec I_c=\tfrac{(P_c-jQ_c)\,a}{V} \]
    \[ I_0=\tfrac13(\vec I_a+\vec I_b+\vec I_c)=B-jC \]
    \[ I_2=\tfrac13(\vec I_a+a^2\vec I_b+a\vec I_c)=\overline{B+jC} \]
    con
    \[ B=\tfrac{1}{3V}(P_a+a^2P_b+aP_c) \]
    \[ C=\tfrac{1}{3V}(Q_a+a^2Q_b+aQ_c) \]
    \column{.48\textwidth}
    \begin{block}{Paralelogramo}
      \footnotesize\[|B-jC|^2+|B+jC|^2=2|B|^2+2|C|^2\]
    \end{block}
    \begin{itemize}\setlength\itemsep{2pt}
      \item Mínimo de $|I_0|^2+|I_2|^2$ con $C=0$: vale $2|B|^2$, \;$|I_0|=|I_2|=|B|$.
      \item Anular $I_0$ con $Q$: $jC=B\Rightarrow |I_2|=2|B|$.
      \item $B=0\iff P_a=P_b=P_c$ (porque $1+a+a^2=0$).
      \item Resultado 1: $I_0=I_2=0\iff B=C=0\iff S_{h,a}=S_{h,b}=S_{h,c}$.
    \end{itemize}
  \end{columns}
  \volver
\end{frame}

\begin{frame}[label=r2]{R2. ¿Por qué no basta un compensador de reactivos?}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
    \begin{itemize}\setlength\itemsep{4pt}
      \item Un STATCOM o un IC con $P=0$ solo mueve $C$ (Resultado 2).
      \item Con $C$ libre: puede llevar $|I_0|$ o $|I_2|$ a cero, \textbf{pero no ambos} si $B\neq0$.
      \item Lo mejor que logra en total: $|I_0|^2+|I_2|^2=2|B|^2$.
      \item Los compensadores clásicos con transferencia de \textbf{potencia activa} entre fases (p.\ ej., Steinmetz, convertidores trifásicos) sí reducen $B$, pero son equipos nuevos y trifásicos.
    \end{itemize}
    \column{.48\textwidth}
    \begin{tikzpicture}
      \node[card=5.8cm] {\textbf{\color{usachT}Lo que aporta la propuesta}\\[2pt] Usa recursos que \textbf{ya existen} (Micro-redes con batería), monofásicos y distribuidos, y les agrega la coordinación para mover $P$ \textbf{en el tiempo} y $Q$ \textbf{sin costo energético}.};
    \end{tikzpicture}

    \vspace{6pt}
    \begin{tikzpicture}
      \node[card=5.8cm,draw=usachO] {\textbf{\color{usachO}Respuesta corta}\\[2pt] ``$Q$ traslada el desbalance entre secuencias; para achicarlo hay que igualar las $P$, y eso requiere energía. Esa energía está en las baterías de las Micro-redes.''};
    \end{tikzpicture}
  \end{columns}
  \volver
\end{frame}

\begin{frame}[label=r3]{R3. El planteamiento anterior y por qué no bastaba}
  \footnotesize
  Ley evaluada: \;$\Pic{i}^{*}=\bar P_{\mathrm{net}}+\frac{E_i}{\tau}\big(\SOCw-\SOC{i}\big)$, \quad $\SOCw=\sum_iE_i\SOC{i}/\sum_iE_i$.

  \vspace{4pt}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
    \textbf{\color{usachT}1. La suma se anula}
    \[\textstyle\sum_iE_i(\SOCw-\SOC{i})=0\]
    \[\textstyle\Rightarrow\;\sum_i\Pic{i}=\sum_iP_{\mathrm{net},i}\;\Rightarrow\;\sum_i\PB{i}\equiv0\]
    El SOC promedio no cambia: las baterías solo se pasan energía entre sí, a través de la red.

    \vspace{4pt}
    \textbf{\color{usachT}2. Régimen permanente}\\
    Con $P_{\mathrm{net},i}$ constante, $\Pic{i}\to P_{\mathrm{net},i}$: el mismo desbalance que sin coordinar. Es un filtro $1/(1+s\tau)$.
    \column{.48\textwidth}
    \textbf{\color{usachT}3. Sin desbalance previo}\\
    Por el Resultado 1, toda transferencia sostenida entre fases exige $P_{h,a}\neq P_{h,b}$: \textbf{crea} desbalance.

    \vspace{6pt}
    \begin{tikzpicture}
      \node[card=5.6cm,draw=usachO] {\textbf{\color{usachO}Qué se rescata:} la idea de usar las Micro-redes por fase, el consenso y el balance de SOC. Ahora el consenso trabaja \textbf{dentro} de cada fase y el objetivo vive en la \textbf{cabecera}.};
    \end{tikzpicture}
  \end{columns}
  \volver
\end{frame}


\begin{frame}[label=r4]{R4. Algoritmo del secundario (por fase)}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{.54\textwidth}
    \textbf{Problema de la fase $\phi$:}
    \begin{align*}
      \min_{\PB{k}}\;&\textstyle\sum_{k\in\Nset{\phi}} \tfrac{1}{2E_k}\PB{k}^2+b_k(\SOC{k})\,\PB{k}\\
      \text{s.a.}\;&\textstyle\sum_{k}\Pic{k}=P^*_\phi+\Delta P_\phi^{\mathrm{corr}},\quad |\PB{k}|\le P_{B,k}^{\max}
    \end{align*}
    $b_k=\beta(\mathrm{SOC}^{\mathrm{ref}}-\SOC{k})$.

    \vspace{4pt}
    \textbf{Iteración} (Yang--Tan--Xu [11]), pesos $w_{kj}$ en $\mathcal G_\phi$:
    \begin{align*}
      \lambda_k&\leftarrow\textstyle\sum_j w_{kj}\lambda_j+\varepsilon\,y_k\\
      \PB{k}&\leftarrow\arg\min\{C_k(\PB{k})-\lambda_k\PB{k}\}\ \text{(con saturación)}\\
      y_k&\leftarrow\textstyle\sum_j w_{kj}y_j-(\PB{k}^{\mathrm{new}}-\PB{k}^{\mathrm{old}})
    \end{align*}
    \column{.44\textwidth}
    \begin{itemize}\setlength\itemsep{3pt}
      \item $y_k$ estima el desajuste de potencia de la fase; su suma se conserva.
      \item El líder inicializa $y$ con la consigna y suma la corrección medida en cabecera.
      \item \textbf{Reactivos:} $\sum_k\Qic{k}=Q^*_\phi$, consenso en la razón de uso $\Qic{k}/\sqrt{(S_k^{\max})^2-\Pic{k}^2}$.
      \item Por estudiar: convergencia con saturación, retardos y pérdida de enlaces.
    \end{itemize}
  \end{columns}
  \volver
\end{frame}
