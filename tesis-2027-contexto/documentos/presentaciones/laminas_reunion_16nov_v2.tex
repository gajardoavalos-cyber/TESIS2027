  \begin{columns}[T,onlytextwidth]
    \column{.325\textwidth}
    \begin{tikzpicture}\node[kc=cT,text width=3.85cm,minimum height=3.0cm]{%
      \kt{cT!85!black}{1. Cuánto margen aparta}\\[2pt]
      Separa el desbalance en dos partes:\\
      $\bullet$ \textbf{previsible} (perfil conocido, dura horas): se aparta como \textbf{energía}, dentro de la banda de SoC;\\
      $\bullet$ \textbf{error} del pronóstico (media cero): se aparta como \textbf{potencia} $r_k$, con margen de 15 min.};\end{tikzpicture}
    \column{.335\textwidth}
    \begin{tikzpicture}\node[kc=cL,text width=3.95cm,minimum height=3.0cm]{%
      \kt{verdeL}{2. Por qué ya no chocan}\\[2pt]
      El terciario \textbf{no} tiene el balance en su objetivo; solo aparta lo que el lazo rápido va a usar. El lazo rápido no cambia la compra total (lo que una fase entrega, las otras lo absorben), así que \textbf{no deshace el plan}.};\end{tikzpicture}
    \column{.325\textwidth}
    \begin{tikzpicture}\node[kc=usachR,text width=3.85cm,minimum height=3.0cm]{%
      \kt{usachR}{3. Si una MR se equivoca}\\[2pt]
      El IC mantiene su consigna y la BESS absorbe la diferencia: \textbf{consume margen}. Por eso el margen incluye el error $\bar\varepsilon_k$ y el terciario replanifica cada 15 min con la energía medida.};\end{tikzpicture}
  \end{columns}
\end{frame}

\begin{frame}{Terciario: objetivo, envolvente que recibe y reserva que asigna}
  \vspace{-0.32cm}
  \begin{columns}[T,onlytextwidth]
    \column{.45\textwidth}
    \begin{tikzpicture}\node[kc=cS,text width=5.5cm]{%
      \kt{usachTd}{1. Envolvente que recibe de cada MR}\\[2pt]
      $\mathcal E_k(t)=\{P^{\min}_k,P^{\max}_k,W^{\min}_k,W^{\max}_k\}(t)$\\[1pt]
      $\bullet$ $P$: cuánto puede entregar o absorber su IC.\\
      $\bullet$ $W$: cuánta energía acumulada puede entregar sin salir del rango de SoC de su BESS y EV.\\
      $\bullet$ La calcula su secundario con un pronóstico simple: perfil conocido $+$ error $\bar\varepsilon_k$.\\[2pt]
      \kt{usachTd}{Y por fase}\\[1pt]
      $\hat D_\phi$: cuánto se aparta la fase $\phi$ del promedio (demanda no gestionada $=P_{h,\phi}+\sum P_{\mathrm{IC},k}$), y su error típico $\sigma_\phi$.};\end{tikzpicture}\\[3pt]
    \begin{tikzpicture}\node[kc=cT,text width=5.5cm]{%
      \kt{cT!85!black}{2. Objetivo: solo energía}\\[2pt]
      \hspace*{6pt}$\displaystyle\min\ \sum_{t}c(t)\,g(t)\,\Delta t$\\[1pt]
      \hspace*{6pt}con\ \ $g\ge \hat P_{L,\mathrm{tot}}-\sum_k P^*_k$\ \ y\ \ $g\ge0$\\[1pt]
      };\end{tikzpicture}

    \column{.535\textwidth}
    \begin{tikzpicture}\node[kc=cT,text width=6.7cm]{%
      \kt{cT!85!black}{3. Reserva (margen) que asigna}\\[2pt]
      Decide para cada MR: plan $P^*_k$, uso previsto del margen $\hat y_k$ y margen para el error $r^{\uparrow}_k,r^{\downarrow}_k$.\\[3pt]
      \textbf{(a)} El margen alcanza para lo que el lazo rápido \textbf{va a usar}:\\[1pt]
      \hspace*{8pt}$\displaystyle\sum_{k\in\mathcal N_\phi}\hat y_k=\hat D_\phi-\big(S_\phi-\bar S\big)-\rho_\phi,\quad |\rho_\phi|\le\bar\rho$\\[1pt]
      \hspace*{8pt}{\color{grisM}lo que el lazo verá en la fase $\phi$, con $S_\phi=\sum_{k\in\mathcal N_\phi}P^*_k$; $\rho_\phi$: residual admisible}\\[3pt]
      \textbf{(b)} Plan $+$ margen caben en la potencia ($\tilde P_k=P^*_k+\hat y_k$):\\[1pt]
      \hspace*{8pt}$P^{\min}_k+r^{\downarrow}_k\ \le\ \tilde P_k\ \le\ P^{\max}_k-r^{\uparrow}_k$\\[3pt]
      \textbf{(c)} Plan $+$ margen caben en la energía (banda de SoC):\\[1pt]
      \hspace*{8pt}$W^{\min}_k+r^{\downarrow}_k\tau+\bar\varepsilon_k\ \le\ \textstyle\sum_{s\le t}\tilde P_k\Delta t\ \le\ W^{\max}_k-r^{\uparrow}_k\tau-\bar\varepsilon_k$\\[3pt]
      \textbf{(d)} Cada fase tiene margen para el error:\\[1pt]
      \hspace*{8pt}$\sum_{k\in\mathcal N_\phi}r^{\uparrow}_k\ge z\sigma_\phi+m,\qquad \sum_{k\in\mathcal N_\phi}r^{\downarrow}_k\ge z\sigma_\phi+m$\\[2pt]
      };\end{tikzpicture}\\[3pt]
    \begin{tikzpicture}\node[kc=cP,text width=6.7cm]{%
      \kt{cP}{Qué entrega}\; A cada IC$_k$: $P^*_k$ y $[r^{\downarrow}_k,r^{\uparrow}_k]$. Al lazo rápido: $R_\phi=\sum r_k$ por fase.};\end{tikzpicture}
  \end{columns}
