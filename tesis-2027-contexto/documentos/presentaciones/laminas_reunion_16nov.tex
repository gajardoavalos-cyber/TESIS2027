\begin{frame}{Terciario: objetivo, envolvente que recibe y reserva que asigna}
  \vspace{-0.32cm}
  \begin{columns}[T,onlytextwidth]
    % ---------------- izquierda: lo que recibe y cómo se dimensiona la reserva
    \column{.445\textwidth}
    \begin{tikzpicture}\node[kc=cS,text width=5.45cm]{%
      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{usachTd}1. Lo que informa cada MR $k$ (envolvente)}\\[2pt]
      $\mathcal E_k(t)=\{P^{\min}_k,\,P^{\max}_k,\,W^{\min}_k,\,W^{\max}_k\}(t)$\\[2pt]
      $\bullet$ \textbf{Potencia}: límites del IC, BESS y EV, ya descontado el pronóstico de FV y carga.\\
      $\bullet$ \textbf{Energía}: $W$ = energía acumulada que entrega el IC. Sus límites vienen de los de BESS y EV (incluye $E_{\mathrm{req}}$ del EV):\\[1pt]
      \hspace*{3pt}$W^{\max}_k(t)=E_k(0)+\!\sum\!(\hat P_{\mathrm{PV}}-\hat P_{L})\Delta t-E^{\min}_k$\\
      \hspace*{3pt}$W^{\min}_k(t)=E_k(0)+\!\sum\!(\hat P_{\mathrm{PV}}-\hat P_{L})\Delta t-E^{\max}_k$\\[1pt]
      $\bullet$ \textbf{Pronóstico simple}: perfil conocido $+$ error acotado $\bar\varepsilon_k$.\\
      {\color{grisM}Ejemplo numérico en la lámina de respaldo.}};\end{tikzpicture}\\[3pt]
    \begin{tikzpicture}\node[kc=cL,text width=5.45cm]{%
      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{verdeL}3. Cuánta reserva pide cada fase}\\[2pt]
      Demanda no gestionada, sin medir clientes: $P_{\mathrm{cl},\phi}=P_{h,\phi}+\sum_{k\in\mathcal N_\phi}P_{\mathrm{IC},k}$\\
      Desviación: $D_\phi=\hat P_{\mathrm{cl},\phi}-\tfrac13\sum_\psi \hat P_{\mathrm{cl},\psi}$\\
      $R^{\uparrow}_\phi=[D_\phi]^{+}+m,\qquad R^{\downarrow}_\phi=[-D_\phi]^{+}+m$\\
      {\color{grisM} la fase cargada pide reserva para entregar; las livianas, para absorber.}};\end{tikzpicture}

    % ---------------- derecha: el problema
    \column{.54\textwidth}
    \begin{tikzpicture}\node[kc=cT,text width=6.7cm]{%
      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{cT!85!black}2. Problema del terciario (cada 15 min, horizonte $H=12$ h)}\\[3pt]
      \textbf{Variables} por MR $k$ y paso $t$: plan $P^*_k$, reserva $r^{\uparrow}_k,r^{\downarrow}_k\ge0$\\[3pt]
      \textbf{Objetivo: solo energía}\\[1pt]
      \hspace*{8pt}$\displaystyle\min\ \sum_{t\in\mathcal T_H} c(t)\,P^{+}_{\mathrm{imp}}(t)\,\Delta t,\qquad P_{\mathrm{imp}}=\hat P_{L,\mathrm{tot}}-\sum_k P^*_k$\\[2pt]
      {\color{grisM}\hspace*{8pt}(con tarifa plana equivale a minimizar la energía comprada)}\\[4pt]
      \textbf{Restricciones}\\[2pt]
      \hspace*{8pt}(a) potencia:\quad $P^{\min}_k+r^{\downarrow}_k\ \le\ P^*_k\ \le\ P^{\max}_k-r^{\uparrow}_k$\\[3pt]
      \hspace*{8pt}(b) energía, con $W^*_k(t)=\sum_{s\le t}P^*_k(s)\,\Delta t$:\\[1pt]
      \hspace*{14pt}$W^{\min}_k+r^{\downarrow}_k\tau+\bar\varepsilon_k\,\le\, W^*_k\,\le\, W^{\max}_k-r^{\uparrow}_k\tau-\bar\varepsilon_k$\\[3pt]
      \hspace*{8pt}(c) reserva por fase:\\[1pt]
      \hspace*{14pt}$\sum_{k\in\mathcal N_\phi}r^{\uparrow}_k\ge R^{\uparrow}_\phi,\qquad \sum_{k\in\mathcal N_\phi}r^{\downarrow}_k\ge R^{\downarrow}_\phi$\\[3pt]
      {\color{grisM}$\tau$: tiempo que debe sostenerse la reserva (p.\ ej.\ 15 min). Problema lineal: óptimo global en segundos.}};\end{tikzpicture}\\[3pt]
    \begin{tikzpicture}\node[kc=cP,text width=6.7cm]{%
      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{cP}4. Qué entrega y qué pasa si una MR se equivoca}\\[2pt]
      $\bullet$ A cada IC$_k$: $P^*_k$ y $[r^{\downarrow}_k,r^{\uparrow}_k]$; al lazo rápido: $R_\phi=\sum r_k$. Replanifica cada 15 min con $E_k$ real.\\
      $\bullet$ Error de pronóstico: el IC mantiene su consigna y la BESS absorbe la diferencia $\Rightarrow$ \textbf{consume reserva de energía} ($\bar\varepsilon_k$ en (b)).};\end{tikzpicture}
  \end{columns}

      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{verdeL} 2. Cómo calcula}\\[2pt]
      $\sum_\phi e_\phi=0\Rightarrow\sum_\phi\Delta P_\phi=0$: mueve potencia entre fases sin cambiar la compra total, así que \textbf{no deshace el plan}.\\
      Ej.: fase $a$ +6 kW $\Rightarrow e=(4,-2,-2)$: $a$ entrega 4 kW, $b$ y $c$ absorben 2 kW c/u.};\end{tikzpicture}
    \column{.325\textwidth}
    \begin{tikzpicture}\node[kc=cP,text width=3.85cm,minimum height=2.05cm]{%
      {\fontsize{6.6}{7.8}\selectfont\bfseries\color{cP} 3. Cómo reparte}\\[2pt]
      $\Delta P_k=u_\phi\,r_k$ (cada IC usa su propia reserva). Aporta en proporción a lo que le asignó el terciario. $|u_\phi|=1$: reserva agotada.\\
      $Q$: no se usa ($R>X$ en BT).};\end{tikzpicture}
